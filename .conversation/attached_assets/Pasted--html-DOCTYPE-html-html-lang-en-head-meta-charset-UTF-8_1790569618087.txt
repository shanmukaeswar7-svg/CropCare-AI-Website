  /* NAVIGATION */

  function showPage(page) {

    document.querySelectorAll(".page")
      .forEach(p => p.classList.remove("active"));

    const target =
      document.getElementById(page);

    if (target) {
      target.classList.add("active");
    }


    document.querySelectorAll(".nav-links button")
      .forEach(btn => btn.classList.remove("active"));

    const nav =
      document.getElementById("nav-" + page);

    if (nav) {
      nav.classList.add("active");
    }


    if (page === "history") {
      renderHistory();
    }

    if (page === "dashboard") {
      renderDashboard();
    }

    window.scrollTo({

  /* IMAGE UPLOAD */

  const imageInput =
    document.getElementById("imageInput");

  imageInput.addEventListener("change", function() {

    const file = this.files[0];

    if (!file) return;

    if (
      ![
        "image/jpeg",
        "image/png",
        "image/webp"
      ].includes(file.type)
    ) {

      alert(
        "Please select a JPEG, PNG or WEBP image."
      );

      return;
    }


    if (file.size > 8 * 1024 * 1024) {

      alert(
        "Image must be smaller than 8 MB."
      );

      return;
    }


    selectedFile = file;

    const reader =
      new FileReader();


    reader.onload = function(e) {

      const preview =
        document.getElementById("preview");

      preview.src =
        e.target.result;

      preview.style.display =
        "block";

      document.getElementById("uploadArea")
        .style.display = "none";

    };


    reader.readAsDataURL(file);

  });


  function removeImage() {

    selectedFile = null;

    imageInput.value = "";

    document.getElementById("preview")
      .style.display = "none";

    document.getElementById("uploadArea")
      .style.display = "flex";
  }

  function capitalize(text) {

    if (!text) return "";

    return text.charAt(0).toUpperCase()
      + text.slice(1);

  }


  /* HISTORY */

  function renderHistory() {

    const list =
      document.getElementById(
        "historyList"
      );


    const search =
      document.getElementById(
        "historySearch"
      ).value
      .toLowerCase();


    const status =
      document.getElementById(
        "historyStatus"
      ).value;


    const severity =
      document.getElementById(
        "historySeverity"
      ).value;


    let filtered =
      diagnosisHistory.filter(
        item => {

          const matchesSearch =
            item.cropName
              .toLowerCase()
              .includes(search)
            ||
            item.diseaseName
              .toLowerCase()
              .includes(search);


          const matchesStatus =
            status === "all"
            ||
            (
              status === "possible"
              &&
              item.diagnosisStatus
                === "possible_disease"
            )
            ||
            (
              status === "healthy"
              &&
              item.diagnosisStatus
                === "healthy"
            )
            ||
            (
              status === "uncertain"
              &&
              item.diagnosisStatus
                === "uncertain"
            );


          const matchesSeverity =
            severity === "all"
            ||
            item.severity === severity;


          return (
            matchesSearch
            &&
            matchesStatus
            &&
            matchesSeverity
          );

        }
      );


    if (filtered.length === 0) {

      list.innerHTML = `

        <div class="card" style="text-align:center;">

          <div style="font-size:50px;">
            🌱
          </div>

          <h2 style="margin-top:15px;">
            No diagnoses found
          </h2>

          <p style="color:#737b73;margin-top:7px;">
            Start by uploading a crop photo.
          </p>

          <button
            class="btn btn-primary"
            style="margin-top:18px;"
            onclick="showPage('diagnose')"
          >
            Diagnose a Crop
          </button>

        </div>

      `;

      return;
    }


    list.innerHTML =
      filtered.map(item => `

        <div
          class="history-item"
          onclick="openHistory('${item.id}')"
          style="cursor:pointer;"
        >

          <div>

            <small style="color:#737b73;">
              ${item.cropName}
            </small>

            <h3>
              ${item.diseaseName}
            </h3>

            <div class="history-meta">
              ${new Date(item.createdAt).toLocaleString()}
            </div>

          </div>

          <div style="display:flex;gap:8px;flex-wrap:wrap;">

            <span class="badge badge-yellow">
              ${item.confidence}%
            </span>

            <span class="badge badge-gray">
              ${capitalize(item.severity)}
            </span>

          </div>

        </div>

      `).join("");

  }


  function openHistory(id) {

    const item =
      diagnosisHistory.find(
        x => x.id === id
      );

    if (!item) return;

    displayResult(item);

    showPage("result");

  }
